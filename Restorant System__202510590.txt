import java.util.*;

public class work {
    public static void main(String[] args) {

// Create Scanner for keyboard input
        Scanner input = new Scanner(System.in);

// Constant: price of each drink
         double DRINK_PRICE = 15.50;

// Collect customer information
        System.out.print("Enter customer's name: ");
        String name = input.nextLine();

        System.out.print("Enter customer's age: ");
        int age = input.nextInt();

        System.out.print("Enter meal price: $`");
        double mealPrice = input.nextDouble();

        System.out.print("Enter number of drinks: ");
        int drinks = input.nextInt();

        System.out.print("Is the customer a student? (true/false): ");
        boolean isStudent = input.nextBoolean();

        System.out.print("Enter service rating (1-5): ");
        int rating = input.nextInt();

// Calculate drinks cost
        double drinksCost = drinks * DRINK_PRICE;

// Calculate subtotal
        double subtotal = mealPrice + drinksCost;

// Calculate discount
        double discountRate = 0.0;

        if (isStudent && age >= 60) {
            discountRate = 0.10;
        } else if (isStudent) {
            discountRate = 0.10;
        } else if (age >= 60) {
            discountRate = 0.05;
        } else {
            discountRate = 0.0;
        }

        double discountAmount = subtotal * discountRate;

// Calculate amount after discount
        double discountedTotal = subtotal - discountAmount;

// Validate service rating and calculate tip
        double tipAmount = 0.0;

        if (rating < 1 || rating > 5) {
            System.out.println("ERROR: Service rating must be between 1 and 5.");
        } else if (rating >= 4 && rating <= 5) {
            tipAmount = discountedTotal * 0.10;
        } else {
            tipAmount = 0.0;
        }

// Calculate exact final bill
        double finalBill = discountedTotal + tipAmount;

// Cast final bill to int
        int wholeAmount = (int) finalBill;

// Print restaurant receipt
        System.out.println("\n========== SMART RESTAURANT ");
        System.out.println("Customer: " + name);
        System.out.printf("Meal cost: $%.2f%n", mealPrice);
        System.out.printf("Drinks cost: $%.2f%n", drinksCost);
        System.out.printf("Discount: $%.2f%n", discountAmount);
        System.out.printf("Tip: $%.2f%n", tipAmount);
        System.out.printf("Final bill: `$%.2f%n", finalBill);
        System.out.println("Whole amount: $" + wholeAmount);
        System.out.println("============================");


    }
}
